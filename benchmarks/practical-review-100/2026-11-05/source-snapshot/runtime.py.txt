def read(p):return json.loads(Path(p).read_text(encoding='utf-8-sig'))
def sha(p):return hashlib.sha256(Path(p).read_bytes()).hexdigest()
def packed(v):return json.dumps(v,ensure_ascii=False,separators=(',',':')).encode('utf-8')
def now():return dt.datetime.now(dt.timezone.utc).isoformat()
def save(p,v):
 p=Path(p);p.parent.mkdir(parents=True,exist_ok=True)
 with p.open('x',encoding='utf-8') as f:json.dump(v,f,ensure_ascii=False,indent=2);f.write('\n');f.flush();os.fsync(f.fileno())
def replace(p,v):
 p=Path(p);q=p.with_suffix('.tmp');q.write_text(json.dumps(v,ensure_ascii=False,indent=2)+'\n',encoding='utf-8');os.replace(q,p)
def load(name,p):
 s=importlib.util.spec_from_file_location(name,p);m=importlib.util.module_from_spec(s);s.loader.exec_module(m);return m
def event(v):
 with (RUN/'journal.jsonl').open('a',encoding='utf-8') as f:f.write(json.dumps({'timestamp':now(),**v},ensure_ascii=False)+'\n');f.flush();os.fsync(f.fileno())
def verify():
 f=read(ROOT/'EXECUTION_FREEZE.json')
 for p,h in f['inputs'].items():assert sha(p)==h,'Frozen input drift: '+p
 assert sha(f['cli'])==f['cli_sha256'] and sha(f['ollama_exe'])==f['ollama_sha256']
 return f
def guard():
 import psutil
 r={'timestamp':now(),'ram_available_bytes':psutil.virtual_memory().available,'disk_free_bytes':shutil.disk_usage(ROOT).free}
 assert r['ram_available_bytes']>=6*2**30 and r['disk_free_bytes']>=5*2**30,'Resource guard STOP'
 return r
def preservation():
 files=read(ROOT/'PRESERVATION_BEFORE.json')['files'];bad=[p for p,h in files.items() if not Path(p).is_file() or sha(p)!=h]
 return {'timestamp':now(),'files_checked':len(files),'passed':not bad,'mismatches':bad}
def workspace_for(f,tid,arm,stage):
 # Shared consultation and normal continuation intentionally share one session;
 # the two delegated conditions, fallbacks and blinded audits never share one.
 return Path(f['actor_root'])/tid/('sol-dialogue' if arm in ['SHARED_DIALOGUE','SOL_CONTINUE'] else arm if arm in ['TERRA_GEMMA_TERRA','LUNA_GEMMA_LUNA','PRACTICAL'] else arm+'/'+stage)
def argv_for(f,model,workspace,out,schema=None,session=None,persist=False,effort='medium'):
 assert model in ['gpt-6.1-sol','gpt-6.1-terra','gpt-6-luna']
 args=[f['cli'],'-C',str(workspace),'--sandbox','read-only','exec','--ignore-user-config','--color','never','--skip-git-repo-check','-c',f'model_reasoning_effort="{effort}"','-c','features.fast_mode=false','-c','agents.enabled=false','-c','features.multi_agent=false','-c','features.multi_agent_v2=false']
 if session:args+=['resume',session,'--ignore-user-config']
 elif not persist:args+=['--ephemeral']
 args+=['--json','-m',model,'--output-last-message',str(out/'answer.txt')]
 if schema is not None:args+=['--output-schema',str(out/'schema.json')]
 return args+['-']
def inspect_events(events):
 allowed={'agent_message','reasoning','todo_list','plan','command_execution'}
 unknown=[e.get('item',{}).get('type') for e in events if e.get('type','').startswith('item.') and e.get('item',{}).get('type') not in allowed]
 commands=[e['item'] for e in events if e.get('type')=='item.completed' and e.get('item',{}).get('type')=='command_execution']
 forbidden=re.compile(r'private|EXPECTED|reference\.py|auth\.json|https?://|Invoke-WebRequest|curl\b|wget\b|run-001|analysis-001|natural-handoff|practical-review|luna-escalation|large-deliverable|review-repair|quality-cost|decomposition|business-500|general-500|work-contract-10',re.I)
 return {'unknown_types':unknown,'commands':commands,'out_of_scope':[c for c in commands if forbidden.search(str(c.get('command','')))],'OS_read_isolation_proven':False}
FIELDS=['input_tokens','cached_input_tokens','output_tokens','reasoning_output_tokens']
def delta(current,previous):
 assert all(type(current.get(k))==int and current[k]>=0 for k in FIELDS),'Invalid cumulative counters'
 d={k:current[k]-previous.get(k,0) for k in FIELDS}
 assert all(v>=0 for v in d.values()),'Non-monotonic session counters'
 assert d['cached_input_tokens']<=d['input_tokens'] and d['reasoning_output_tokens']<=d['output_tokens'],'Invalid token subsets'
 return d
def price(u,model,rate):
 r=rate['rates'][model]
 return ((u['input_tokens']-u['cached_input_tokens'])*Decimal(r['input'])+u['cached_input_tokens']*Decimal(r['cached_input'])+u['output_tokens']*Decimal(r['output']))/r['denominator']
