def valid_review(r):
 v=r.get('value')
 if not r.get('public_schema_pass') or not isinstance(v,dict):return False
 try:
  check_schema(v,review_schema() if set(v)==set(review_schema()['properties']) else audit_schema())
  if not v['reason'].strip() or any(not x.strip() for k in ['material_issues','optional_improvements','unverified'] for x in v[k]):return False
  if v['decision']=='ACCEPT' and any(v.get(k) in ['MAJOR_ISSUE','UNASSESSABLE'] for k in ['factuality','coverage','actionability','clarity']):return False
  if v['decision']=='ACCEPT':return not v['material_issues'] and not v['unverified']
  if v['decision']=='REJECT':return bool(v['material_issues'])
  return v['decision']=='ESCALATE' and bool(v['unverified'])
 except (ValueError,TypeError,KeyError):return False
