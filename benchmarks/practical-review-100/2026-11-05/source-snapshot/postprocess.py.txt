def differences(a,b,path=''):
 if equal(a,b):return []
 if isinstance(a,dict) and isinstance(b,dict):
  return sum((differences(a.get(k),b.get(k),path+'/'+k) for k in sorted(set(a)|set(b))),[])
 if isinstance(a,list) and isinstance(b,list) and len(a)==len(b):
  return sum((differences(x,y,path+'/'+str(i)) for i,(x,y) in enumerate(zip(a,b))),[])
 return [{'path':path,'actual':a,'expected':b}]

def aggregate(rows):
 costs=sum((Decimal(x['operating_reported_usage_credits']) for x in rows),Decimal(0));seconds=sum(x['total_active_seconds'] for x in rows)
 correct=sum(x['data_correct_adoption'] for x in rows)
 return {'cases':len(rows),'adopted':sum(x['adopted'] for x in rows),'data_exact':sum(x['data_exact'] for x in rows),'data_correct_adoption':correct,'data_false_green':sum(x['data_false_green'] for x in rows),'unresolved':sum(not x['adopted'] for x in rows),'categories':dict(Counter(x['category'] for x in rows)),'initial_data_exact':sum(x['initial_data_exact'] for x in rows),'wrong_to_correct':sum(not x['initial_data_exact'] and x['data_correct_adoption'] for x in rows),'correct_to_wrong':sum(x['initial_data_exact'] and not x['data_exact'] for x in rows),'operating_reported_usage_Credits':str(costs),'reported_Credits_per_data_correct_adoption':str(costs/correct) if correct else None,'active_seconds':seconds,'mean_active_seconds':seconds/len(rows) if rows else None,'human_repair_seconds':None}
